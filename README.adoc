A set of kotlin extensions

* JavaDoc Generation

https://github.com/Kotlin/dokka[Dokka Gradle Plugin]
[source,groovy]
[subs="attributes"]
----
plugins {
  id("org.jetbrains.dokka-javadoc") version "2.0.0"
}

tasks.register<Jar>("javadocJar") {
    archiveClassifier.set("javadoc")
    destinationDirectory.set(layout.buildDirectory.dir("libs"))
    from(layout.buildDirectory.dir("dokka/javadoc"))
    dependsOn("dokkaGeneratePublicationJavadoc")
}
----

* GPG Generation

https://docs.gradle.org/current/userguide/signing_plugin.html[Signing Gradle Plugin]

[source,groovy]
[subs="attributes"]
----
plugins {
  id("signing")
}

ext["signing.gnupg.executable"]="gpg"
ext["signing.gnupg.homeDir"]="/Users/adrian/.gnupg"
ext["signing.gnupg.keyName"]=System.getenv("GPG_KEYNAME")
ext["signing.gnupg.passphrase"]=System.getenv("GPG_PASSPHRASE")

signing {
    useGpgCmd()
    sign(publishing.publications)
}
----
